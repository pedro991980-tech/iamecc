import 'package:flutter/material.dart';

class WaveVisualizer extends StatelessWidget {
  final bool isActive;

  const WaveVisualizer({super.key, required this.isActive});

  @override
  Widget build(BuildContext context) {
    return Container(
      height: 200,
      decoration: BoxDecoration(
        color: const Color(0xFF161B22),
        borderRadius: BorderRadius.circular(24),
        border: Border.all(
          color: isActive ? const Color(0xFF00E5FF) : Colors.white12,
          width: 1.5,
        ),
      ),
      child: Center(
        child: isActive
            ? const Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  CircularProgressIndicator(color: Color(0xFF00E5FF)),
                  SizedBox(width: 16),
                  Text("Campionamento audio in corso...", style: TextStyle(color: Color(0xFF00E5FF)))
                ],
              )
            : const Icon(
                Icons.mic_none_rounded,
                size: 64,
                color: Colors.white38,
              ),
      ),
    );
  }
}
