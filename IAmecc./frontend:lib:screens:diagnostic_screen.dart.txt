import 'package:flutter/material.dart';
import '../widgets/wave_visualizer.dart';

class DiagnosticScreen extends StatefulWidget {
  const DiagnosticScreen({super.key});

  @override
  State<DiagnosticScreen> createState() => _DiagnosticScreenState();
}

class _DiagnosticScreenState extends State<DiagnosticScreen> {
  bool _isListening = false;
  String _statusText = "Pronto per la scansione acustica";

  void _toggleListening() {
    setState(() {
      _isListening = !_isListening;
      _statusText = _isListening ? "Analisi delle frequenze in corso..." : "Pronto per la scansione acustica";
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('IAmecc Diagnostic Studio', style: TextStyle(fontWeight: FontWeight.bold, fontSize: 18)),
        backgroundColor: Colors.transparent,
        elevation: 0,
        centerTitle: true,
      ),
      body: Padding(
        padding: const EdgeInsets.all(24.0),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            // Visualizzatore d'onda acustica
            WaveVisualizer(isActive: _isListening),
            const SizedBox(height: 40),
            
            // Stato corrente
            Text(
              _statusText,
              style: const TextStyle(fontSize: 16, color: Colors.white70),
              textAlign: TextAlign.center,
            ),
            const SizedBox(height: 60),

            // Pulsante di avvio/arresto scansione
            SizedBox(
              width: double.infinity,
              height: 56,
              child: ElevatedButton(
                onPressed: _toggleListening,
                style: ElevatedButton.styleFrom(
                  backgroundColor: _isListening ? Colors.redAccent : const Color(0xFF00E5FF),
                  foregroundColor: _isListening ? Colors.white : Colors.black,
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(16),
                  ),
                  elevation: 8,
                ),
                child: Text(
                  _isListening ? 'INTERROMPI ANALISI' : 'AVVIA DIAGNOSI ACUSTICA',
                  style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}
